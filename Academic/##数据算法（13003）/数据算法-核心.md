```shell

	插入(可用n+1)(O(n))

#🚀顺序表
- init/clear(Seq *s)
- isEmpty/isFull/len
- insert
	isFull?  pos?
	for(i=len;;)  s[i]=s[i-1]
    s[i]=n 
    s.n++
- remove
	isEmpty?  pos?
	*n=s[i]
	for(i=pos;;)  s[i]=s[i+1]
	s.n--
- reverse
	isEmpty? > pos?
	for(i=0;;) swap(&s[i],&s[n-i-1])
- swap
	temp=*x
	*x=*y
	*y=temp

	while(temp!=NULL||pos>0)

	temp.next=curr.next

	curr.next=curr.next.next
	free(temp)

	front,rear=*head
```
